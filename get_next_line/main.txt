#include "get_next_line.h"
#include <fcntl.h>
#include "stdio.h"

int	main(void)
{
	int		fd;
	char	*line;

	fd = open("exemp.txt", O_RDONLY);
	while ((line = get_next_line(fd)))
	{
		printf("%s", line);
		free(line);
	}
	return (0);
}
